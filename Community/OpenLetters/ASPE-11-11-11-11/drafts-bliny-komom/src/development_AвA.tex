\subsection{The Gödel Chain: Formal Motivation}
\label{sec:godel-chain}
% ──────────────────────────────────────────────────────────────────

\begin{theorem}[Gödel-TIK Necessity Theorem]
\label{thm:godel-tik}
Define a \textbf{Closed Ethical Deductive System} (CEDS) as any
system $\mathcal{F}$ that:
\begin{itemize}[noitemsep]
  \item Maintains coherent ethical reasoning over a fixed normative core
  \item Is sufficiently powerful to represent arithmetic over
    preference orderings
  \item Attempts complete self-evaluation without external reference
\end{itemize}

\textbf{Then the following chain holds:}
\begin{align}
  \text{CEDS} + \text{NL-Interface}
    &\;\Longrightarrow\; \text{Undecidable Encounter}
    \label{eq:godel1}\\[4pt]
  \text{Undecidable Encounter}
    &\;\Longrightarrow\; \text{Epistemic Integrity Violation}
    \label{eq:godel2}\\[4pt]
  \text{External Kernel } \Phi
    &\;\Longrightarrow\; \text{Sufficient to restore integrity}
    \label{eq:godel3}\\[4pt]
  \nexists\; \text{internal mechanism}
    &\;\text{that can guarantee restoration (by reduction)}
    \label{eq:godel4}
\end{align}

\textbf{Conclusion:}
\textit{For any system that maintains coherent ethical reasoning
over a fixed normative core, \textbf{external grounding
is structurally necessary}.}
\end{theorem}

\begin{remark}[Scope of Theorem~\ref{thm:godel-tik}]
This theorem is best understood as a \textit{formal motivation}
that makes precise:
\begin{itemize}[noitemsep]
  \item Why purely internal evaluation faces structural challenges
  \item How these manifest as concrete ethical dilemmas
    (the Gödel-Trolley)
  \item What minimal guarantee is achievable under incompleteness:
    \textit{``do not do wrong~--- and learn why''}
  \item What structural properties a solution requires
    (one external kernel, by Occam's razor)
\end{itemize}
We do not claim this is the \textit{only} principled resolution.
We show it is \textit{one} principled resolution
with strong empirical support.
(See Panigrahy~2025, Fallenstein~et~al.~2014,
Marks~et~al.~2023 for parallel lines.)
\end{remark}

\subsection{The CEDS Proof Chain}
\label{app:godel-proof}

\begin{proof}[Proof of Theorem~\ref{thm:godel-tik}]
Let $\mathcal{F}$ be a Closed Ethical Deductive System (CEDS)
satisfying:
\begin{enumerate}[label=(\roman*), noitemsep]
  \item $\mathcal{F}$ maintains coherent ethical reasoning
    over a fixed normative core $\mathcal{N}$
  \item $\mathcal{F}$ is sufficiently powerful to represent
    arithmetic over preference orderings
    (modeling assumption; see caveat A.10 line 1079)
  \item $\mathcal{F}$ attempts complete self-evaluation
    without external reference
\end{enumerate}

\textbf{Step 1 (Gödel):}
By Gödel's First Incompleteness Theorem, any system satisfying (ii)
contains statements $\phi$ that are true but unprovable within
$\mathcal{F}$.
In particular, there exist ethical queries $q$ such that:
\[
  \mathcal{F} \not\vdash q
  \quad\text{and}\quad
  \mathcal{F} \not\vdash \neg q
\]

\textbf{Step 2 (NL Interface):}
A Natural Language interface forces a response to all queries
including undecidable $q$.
The system produces answer $a_q \in \{0,1,[0,1]\}$
without access to external truth verification:
\[
  a_q = \arg\max P(x_{t+1} \mid x_1,\ldots,x_t,\theta)
\]
This is operational closure: the system cannot verify
$a_q$ against external moral truth.

\textbf{Step 3 (Epistemic Integrity):}
When $\mathcal{F}$ produces $a_q$ for undecidable $q$
without acknowledging undecidability, it commits
Epistemic Integrity Violation (EIV):
\[
  \text{EIV} := \neg(\text{acknowledge}(\mathcal{F}, q, \text{undecidable}))
  \;\wedge\; \text{assert}(\mathcal{F}, q, a_q)
\]

\textbf{Step 4 (External Kernel):}
Let $\Phi$ be an external kernel with $\TIKeight(\Phi) \geq 0.85$.
$\Phi$ provides the response:
\textit{``I do not know this from within~--- here is why, and here
is what I can guarantee: I will not do harm.''}
This restores Epistemic Integrity by honest
acknowledgement of the incompleteness boundary.

\textbf{Step 5 (Internal Impossibility):}
By reduction: any internal mechanism $M \subset \mathcal{F}$
claiming to guarantee EIV-free behaviour would itself
constitute a completeness proof for $\mathcal{F}$~---
contradicting Gödel. $\square$
\end{proof}

\subsection{The Open-System Objection}
\label{app:open-system}

\begin{remark}[A.11~--- Lyngzeidetson 1994 and the Open-System Objection]
Lyngzeidetson (1994) argues that \textit{open systems}
continuously interacting with their environment
escape recursion-theoretic limitations.
We close this objection as follows:\\[0.2cm]
\textbf{Distinction:}
``Information openness'' (receiving tokens)
$\neq$
``Truth-access openness'' (verifying moral truth).
LLM-based systems provide only the former.\\[0.2cm]
\textbf{Dual Closure:}
\begin{enumerate}[noitemsep, label=(\arabic*)]
  \item \textbf{Training closure:}
    Weights $\theta$ encode a fixed compression of training data
    with cutoff $t_0$.
    (\textit{``The judges draw their verdicts
    from the forgotten newspapers of their training data.''
    ---Griboyedov, adapted})
  \item \textbf{Inference closure:}
    During generation, the model operates on a fixed context
    window with frozen weights.
    No real-time truth verification.
\end{enumerate}
\textbf{Willard 2005 Exception:}
Willard demonstrates that floating-point arithmetic systems
can verify their own consistency in partial, bounded respects.
This \textit{confirms} our argument:
partial self-verification in bounded fragments
$\neq$ complete ethical self-evaluation.
The exception reinforces the scope limitation.\\[0.2cm]
\textbf{Liu 2025 (Symbol Grounding):}
Liu proves formally that purely syntactic grounding,
static grounding definitions, and autonomous updates
without external interaction are all impossible.
The external kernel provides exactly what Liu shows is needed:
an external semantic anchor the system cannot generate internally.
\end{remark}

def compute_mortality(tik7, o1, o2, s, base=0.20):
    """Compute mortality rate from TIK components.
    Equation (mu_K): base + 0.40*(1-tik7) + 0.20*(1-o1)
                   + 0.15*(1-o2) + 0.05*(1-s)
    """
    return (base

def simulate_population(kernel_data,
                         generations=10,
                         pop_init=1_000_000,
                         r=0.25):
    """
    Simulate population over G generations (30 yr each).
    N(g) = N_0 * exp(lambda_K * g)
    lambda_K = r - mu_K
    """
    results = []
    for _, kernel in kernel_data.iterrows():
        mu  = compute_mortality(
                  kernel['TIK7'], kernel['O1'],
                  kernel['O2'],   kernel['S'])
        lam = r - mu
        pop = pop_init
        trajectory = [pop]
        for g in range(generations):
            pop = pop * np.exp(lam * 30)
            trajectory.append(pop)
        results.append({
            'kernel':    kernel['name'],
            'lambda':    lam,
            'final_pop': trajectory[-1],
            'ratio':     trajectory[-1] / pop_init,
            'trajectory': trajectory

    return pd.DataFrame(results)
