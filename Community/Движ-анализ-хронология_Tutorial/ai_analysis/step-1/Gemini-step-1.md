import pandas as pd
import re
from datetime import datetime

# Re-read df2 with proper handling
df1 = pd.read_csv('chronology-process-1.csv')
df2 = pd.read_csv('chronology-process-2.csv', header=None)
df2.columns = df1.columns

# Clean up df1 (remove rows with all NaNs if any)
df1 = df1.dropna(subset=['date', 'event']).reset_index(drop=True)
df2 = df2.dropna(subset=['date', 'event']).reset_index(drop=True)

# Combine data
df_combined = pd.concat([df1, df2], ignore_index=True)

# Function to parse the first date from a string or range
def parse_date(date_str):
    if pd.isna(date_str):
        return None
    # Extract first date pattern dd.mm.yyyy
    match = re.search(r'(\d{2})\.(\d{2})\.(\d{4})', str(date_str))
    if match:
        day, month, year = match.groups()
        return datetime(int(year), int(month), int(day))
    # Handle "до dd.mm.yyyy"
    match_do = re.search(r'до (\d{2})\.(\d{2})\.(\d{4})', str(date_str))
    if match_do:
        day, month, year = match_do.groups()
        return datetime(int(year), int(month), int(day))
    return None

df_combined['parsed_date'] = df_combined['date'].apply(parse_date)
df_combined = df_combined.sort_values('parsed_date').reset_index(drop=True)

# Inspect unique events to find stages/phases
print("Unique events/stages:")
print(df_combined['event'].unique()[:20])

# Current date for future/planned filtering
current_date = datetime(2026, 3, 4)

def is_future(row):
    return row['parsed_date'] > current_date if row['parsed_date'] else False

df_combined['is_future'] = df_combined.apply(is_future, axis=1)

# Look for stages in 'event' or 'what'
def identify_phase(row):
    text = f"{row['event']} {row['what']}"
    match = re.search(r'Stage [A-Z]', text)
    if match:
        return match.group(0)
    # Some phases might be named differently
    if "Baseline" in text: return "Stage A"
    if "Radar" in text or "Послание" in text: return "Stage B"
    return None

df_combined['phase'] = df_combined.apply(identify_phase, axis=1)
# Forward fill phases if they are sequential
df_combined['phase'] = df_combined['phase'].ffill()

print(df_combined[['date', 'event', 'phase', 'is_future']].head(20))
