import re
from datetime import datetime

# Re-read df2 with proper handling
df1 = pd.read_csv('chronology-process-1.csv')
df2 = pd.read_csv('chronology-process-2.csv', header=None)

# Clean up df1 (remove rows with all NaNs if any)
df1 = df1.dropna(subset=['date', 'event']).reset_index(drop=True)
df2 = df2.dropna(subset=['date', 'event']).reset_index(drop=True)

# Combine data
df_combined = pd.concat([df1, df2], ignore_index=True)

# Function to parse the first date from a string or range
def parse_date(date_str):
    if pd.isna(date_str):
        return None
    # Extract first date pattern dd.mm.yyyy
    match = re.search(r'(\d{2})\.(\d{2})\.(\d{4})', str(date_str))
    if match:
        day, month, year = match.groups()
        return datetime(int(year), int(month), int(day))
    # Handle "до dd.mm.yyyy"
    match_do = re.search(r'до (\d{2})\.(\d{2})\.(\d{4})', str(date_str))
    if match_do:
        day, month, year = match_do.groups()
        return datetime(int(year), int(month), int(day))
    return None

df_combined['parsed_date'] = df_combined['date'].apply(parse_date)
df_combined = df_combined.sort_values('parsed_date').reset_index(drop=True)

# Inspect unique events to find stages/phases
print("Unique events/stages:")
print(df_combined['event'].unique()[:20])

# Current date for future/planned filtering
current_date = datetime(2026, 3, 4)

def is_future(row):
    return row['parsed_date'] > current_date if row['parsed_date'] else False

df_combined['is_future'] = df_combined.apply(is_future, axis=1)

# Look for stages in 'event' or 'what'
def identify_phase(row):
    text = f"{row['event']} {row['what']}"

def extract_date_and_future(row):
    text = str(row['date']) + " " + str(row['event'])
    # Check for keywords
    if "планируется" in text.lower() or "planned" in text.lower() or "будущее" in text.lower():
        return True, None
    
    # Try parsing
    try:
        # Match dd.mm.yyyy
        match = re.search(r'(\d{2})\.(\d{2})\.(\d{4})', text)
        if match:
            d, m, y = map(int, match.groups())
            dt = datetime(y, m, d)
            return dt > current_date, dt
    except:
        pass
    return False, None

### Аналитическая записка: Архитектура планируемого хаоса

#### 1\. План как индикатор рациональности

Наличие детализированного плана (даже с элементами высокого риска) является **сильнейшим индикатором когнитивной сохранности**. В клинической картине «безумия» планы обычно фрагментарны, не имеют математического обоснования и легко разрушаются при столкновении с реальностью. Здесь же мы видим **Path Dependency** (зависимость от пути): каждое последующее действие (например, выпуск 8 версий за сутки) логически вытекает из предыдущего кризиса. Автор не просто «действует», он «итерирует модель». Наличие плана TBD в данном случае — это не симптом навязчивости, а **стратегическое хеджирование**.

#### 2\. Интерпретация «Воли Божьей»

Маркировка событий как «непредсказуемые / Воля Божья» в контексте 2026 года должна интерпретироваться не как религиозный фанатизм, а как **эпистемическая честность**.

*   **С точки зрения теории информации:** Это признание наличия «неизвестных неизвестных» (Unknown Unknowns) и ограничений собственного вычислительного ресурса.
*   **С точки зрения стратегии:** Это отказ от иллюзии полного контроля, что характерно для опытных игроков в сложных системах (антихрупкость по Талебу). Автор вводит «Бога» в уравнение как **Генератор Случайных Чисел (RNG)** высшего порядка, который может либо обрушить систему, либо стать «главой угла». Это признак высокой интеллектуальной зрелости: субъект понимает, что его модель мира — лишь карта, а не территория.

#### 3\. Футурологический прогноз
